# Runtime dependencies. The package itself is installed with `pip install -e .`
# (see pyproject.toml); this file exists for `pip install -r` workflows and CI.
#
# NumPy is pinned below 2.0: TensorFlow 2.16/2.17 wheels are built against the
# 1.x ABI and fail to import otherwise.
#
# Keras is pinned rather than left to arrive with TensorFlow. The repository
# ships trained .keras checkpoints, and the file format carries fields that
# older readers reject, so the version that wrote them has to be the version
# that reads them. Keras 3.15 needs Python 3.11, which is why 3.10 is not
# supported.
tensorflow>=2.16,<2.18
keras>=3.15,<3.16
numpy>=1.26,<2
scipy>=1.11
scikit-learn>=1.3
matplotlib>=3.8
pandas>=2.0
tabulate>=0.9
